# 阶段 3：k8s 清单与集群验证 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给出一套可直接交付平台的 k8s 清单（每用户一个 runtime Deployment、知识库管理员的 kb-import Deployment、NAS PVC、NetworkPolicy、Secret/ConfigMap 样例），并在 Mac 的 OrbStack Kubernetes 上跑通 spec §12 的六项验证。

**Architecture:** 清单分三层：`k8s/base/`（命名空间、NAS PVC、ConfigMap、角色表、NetworkPolicy，一次性）、`k8s/templates/`（按工号渲染的 runtime / kb-import Deployment + Service + 令牌 Secret）、`k8s/local/`（只在 Mac 上用的 hostPath PV）。`scripts/k8s/provision.py` 扮演网关的「按工号建 Pod」那一步：读角色表（mock 的 AD 组）、渲染模板、`kubectl apply`；`scripts/k8s/verify-cluster.sh` 跑六项验证。工号与分组按 user 的要求自己 Mock：`k8s/base/configmap-agent-roles.yaml` 里列谁是 `kb-admins`，令牌从本机 `~/.gdaa/grmp.env` 造成每人一个 Secret。

**Tech Stack:** OrbStack Kubernetes v1.35（上下文 `orbstack`，单节点，与 docker 共享镜像存储与 Mac 文件系统）、`kubectl`（`/usr/local/bin/kubectl`）、Python 3（渲染模板，不依赖 envsubst）、阶段 2 的镜像 `gaussdb-agent-runtime:dev` / `gaussdb-agent-kb-import:dev`。

**Spec:** `docs/specs/2026-09-15-k8s-containerization-design.md` §4、§6、§7、§8、§11、§12

## Global Constraints

- 所有 `kubectl` 命令显式带 `--context orbstack`；**绝不** `kubectl config use-context`（user 的默认上下文是另一个项目的 k3s 集群 `opendb-dsh`）。
- 命名空间固定 `gaussdb-agent`；对象命名：`runtime-<工号>`、`kb-import-<工号>`、Secret `grmp-token-<工号>`、`pod-auth-<工号>`。
- Mac 上 NAS 用 hostPath `/tmp/agent-nas-k8s` 模拟（已验证 Pod 内 uid 1000 可读写）；客户环境换成 NFS PV，清单里 PVC 不变。
- 镜像用本机 `:dev`，`imagePullPolicy: IfNotPresent`；客户环境改成仓库地址与 `agent-v0.2-oc1.18.27`。
- GRMP mock 只绑 127.0.0.1：验证期间用 `scripts/tcp-forward.py 8781 8779`，Pod 走 `host.docker.internal:8781`（已验证可达）。
- NetworkPolicy 随清单交付，但 OrbStack 的 CNI 不一定强制执行；本阶段只验证清单合法（`--dry-run=server`），强制生效在客户集群（Calico/Cilium）验证，写进文档。
- Secret 样例文件只含占位符；真实令牌只在运行时由脚本从 `~/.gdaa/grmp.env` 读入 `kubectl create secret`，不落盘、不进仓库。

  - 渲染模板 → `kubectl apply -f -`，等 Deployment Available（默认 120 秒）；
  - `--delete`：删 Deployment/Service/Secret（PVC 与 NAS 目录**不删**）。

```python
#!/usr/bin/env python3
"""网关「按工号建 Pod」那一步的参考实现(spec §8),也是 Mac 集群验证的工具。

  provision.py u1001 --auto-role            # 读角色表决定给不给 kb-import
  provision.py u2001 --kb-admin             # 强制加 kb-import
  provision.py u1001 --delete               # 回收 Deployment/Service/Secret(PVC 与 NAS 目录不动)

令牌只在内存里经 stdin 交给 kubectl,不落盘、不进日志。
"""
from __future__ import annotations

import argparse
import pathlib
import secrets
import string
import subprocess
import sys
import time
from typing import List, Optional

ROOT = pathlib.Path(__file__).resolve().parents[2]
TEMPLATES = ROOT / "k8s" / "templates"
NS = "gaussdb-agent"


def is_kb_admin(roles_text: str, user_id: str) -> bool:
    return user_id in {ln.strip() for ln in roles_text.splitlines() if ln.strip()}


def plan(user_id: str, roles_text: str, auto_role: bool, kb_admin_flag: bool) -> List[str]:
    out = ["runtime"]
    if kb_admin_flag or (auto_role and is_kb_admin(roles_text, user_id)):
        out.append("kb-import")
    return out


def render(template_text: str, **vars_) -> str:
    return string.Template(template_text).substitute(vars_)     # 缺占位符 → KeyError,宁可炸也不要渲染出空值


def _kubectl(ctx: str, *args: str, stdin: Optional[str] = None, check: bool = True) -> subprocess.CompletedProcess:
    return subprocess.run(["kubectl", "--context", ctx, "-n", NS, *args], input=stdin, text=True,
                          capture_output=True, check=check)


def _read_token(env_file: pathlib.Path) -> str:
    for ln in env_file.read_text(encoding="utf-8").splitlines():
        if ln.startswith("GRMP_AUTH_TOKEN="):
            return ln.split("=", 1)[1].strip().strip('"')
    raise SystemExit("令牌文件里没有 GRMP_AUTH_TOKEN=:%s" % env_file)


def _apply_secret(ctx: str, name: str, key: str, value: str) -> None:
    yaml = "apiVersion: v1\nkind: Secret\nmetadata:\n  name: %s\n  namespace: %s\ntype: Opaque\nstringData:\n  %s: %s\n" % (name, NS, key, value)
    _kubectl(ctx, "apply", "-f", "-", stdin=yaml)


def _roles(ctx: str) -> str:
    r = _kubectl(ctx, "get", "configmap", "agent-roles", "-o", "jsonpath={.data.kb-admins}", check=False)
    return r.stdout if r.returncode == 0 else ""


def provision(user_id: str, ctx: str, tag: str, pull_policy: str, kinds: List[str], token_file: Optional[pathlib.Path], wait_s: int) -> None:
    if "runtime" in kinds:
        if token_file is None:
            raise SystemExit("runtime 需要本人 GRMP 令牌:--token-env-file")
        _apply_secret(ctx, "grmp-token-%s" % user_id, "GRMP_AUTH_TOKEN", _read_token(token_file))
    _apply_secret(ctx, "pod-auth-%s" % user_id, "OPENCODE_SERVER_PASSWORD", secrets.token_hex(24))
    for kind in kinds:
        text = render((TEMPLATES / f"{kind}.yaml").read_text(encoding="utf-8"), USER_ID=user_id,
                      IMAGE="gaussdb-agent-%s:%s" % (kind, tag), IMAGE_PULL_POLICY=pull_policy,
                      SUBPATH_ROOT="admins" if kind == "kb-import" else "users")
        _kubectl(ctx, "apply", "-f", "-", stdin=text)
        print("已创建 %s-%s" % (kind, user_id))
    for kind in kinds:
        name = "%s-%s" % (kind, user_id)
        deadline = time.time() + wait_s
        while time.time() < deadline:
            r = _kubectl(ctx, "get", "deploy", name, "-o", "jsonpath={.status.availableReplicas}", check=False)
            if r.stdout.strip() == "1":
                print("%s 就绪" % name)
                break
            time.sleep(3)
        else:
            raise SystemExit("%s %d 秒内未就绪:kubectl --context %s -n %s describe deploy %s" % (name, wait_s, ctx, NS, name))


def deprovision(user_id: str, ctx: str) -> None:
    for kind in ("runtime", "kb-import"):
        _kubectl(ctx, "delete", "deploy,svc", "%s-%s" % (kind, user_id), "--ignore-not-found", "--wait=true", check=False)
    for s in ("grmp-token-%s" % user_id, "pod-auth-%s" % user_id):
        _kubectl(ctx, "delete", "secret", s, "--ignore-not-found", check=False)
    print("已回收 %s 的 Pod/Service/Secret(NAS 目录保留)" % user_id)


def main(argv: Optional[List[str]] = None) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("user_id")
    ap.add_argument("--context", default="orbstack")
    ap.add_argument("--image-tag", default="dev")
    ap.add_argument("--pull-policy", default="IfNotPresent")
    ap.add_argument("--kb-admin", action="store_true")
    ap.add_argument("--auto-role", action="store_true", help="读 ConfigMap agent-roles 决定是否给 kb-import")
    ap.add_argument("--token-env-file", default=str(pathlib.Path.home() / ".gdaa" / "grmp.env"))
    ap.add_argument("--wait-s", type=int, default=120)
    ap.add_argument("--delete", action="store_true")
    a = ap.parse_args(argv)
    if a.delete:
        deprovision(a.user_id, a.context)
        return 0
    kinds = plan(a.user_id, _roles(a.context) if a.auto_role else "", a.auto_role, a.kb_admin)
    provision(a.user_id, a.context, a.image_tag, a.pull_policy, kinds, pathlib.Path(a.token_env_file).expanduser(), a.wait_s)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

**顺带发现的检索缺口（技能层，不是 Pod 问题）**：runtime 模型用现象语言提问「批量开户 insert 等待序列取号」时 `kb.py query` 报「无相似案例」，模型是靠 `grep -ril 序列 /nas/kb/cases` 再 `read` 案例文件才答对的。复现：文件模式下，纯中文的现象查询对**所有**既有案例都不命中（「空闲事务持锁阻塞会话」「两个过程更新顺序相反死锁频发」……全部「无相似案例」），带标识符的查询（`cbst.seq_acct_no`、`autovacuum`、`exchange`）才命中。原因在 `common/kb/query.py::is_strong_token`：强 token = 含 `_ . :` 的标识符或 ≥6 字符且不在泛词表里的英文词；中文按二元组切分永远是 2 字，`insert` / `update` 在泛词表里、`cache` 只有 5 字符——于是不带标识符的中文提问过不了 `_relevant` 的「至少 1 个强 token」门槛。向量库模式下向量相似度 ≥0.45 能绕过这道门，文件模式没有向量，缺口是结构性的。`kb.py search` 不搜 `cases/`（只搜 rules / guides / errata）是设计如此。处理归属：gh_skill 与本仓库同改（开发规范第 2 条），待拍板。
